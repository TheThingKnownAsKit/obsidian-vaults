```Racket
(define parse-myexpression
  (lambda (expr)
    (cond
      ((number? expr) (lit-exp expr))
      ((pair? expr)
       (cond
         ((and (or (eqv? (car expr) '+) (eqv? (car expr) '-))
               (null? (cddr expr)))
          (eopl:error `parse-myexpression "~s has too many arguments" expr)
         )
         ((eqv? (car expr) '+)
          (primapp-exp (add-prim) (parse-myexpression (cadr expr)) (map parse-myexpression (cddr expr))) ; Modified for homework 3, works with list of expr
          )
         ((eqv? (car expr) '-)
          (primapp-exp (subtract-prim) (parse-myexpression (cadr expr)) (map parse-myexpression (cddr expr))) ; Modified for homework 3, works with list of expr
          )
         ((eqv? (car expr) '*)
          (primapp-exp (multiply-prim) (parse-myexpression (cadr expr)) (parse-myexpression (caddr expr))))
         ((eqv? (car expr) '/)
          (primapp-exp (divide-prim) (parse-myexpression (cadr expr)) (parse-myexpression (caddr expr))))
         ((eqv? (car expr) 'add1) ; Modified for homework 3
          (add1-exp (parse-myexpression (cadr expr)))
          )
         ((eqv? (car expr) 'sub1) ; Modified for homework 3
          (sub1-exp (parse-myexpression (cadr expr)))
          )
         (else
          (eopl:error `parse-myexpression "~s is not a sentence" expr)
          )
         )
       )
      (else (eopl:error `parse-myepression "~s is not a sentence" expr))
      )
    )
  )
```

```Racket
(define eval-primapp
  (lambda (prim exp1 exp-list)
    (cases primitive prim
      (add-prim ()
                (+ (eval-expression exp1)
                   (apply + (map eval-expression exp-list))))
      (subtract-prim ()
                     (- (eval-expression exp1)
                        (apply + (map eval-expression exp-list)))) ; Modified for homework 3
      (multiply-prim ()
                     (* (eval-expression exp1)
                        (apply * (map eval-expression exp-list)))) ; Modified for homework 3
      (divide-prim ()
                   (let ((denominator (apply * (map eval-expression exp-list))))
                     (if (zero? denominator)
                         (eopl:error 'eval-primapp "Division by zero")
                     (apply / (map eval-expression exp-list)))))
      )))
```
